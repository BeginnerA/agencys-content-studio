# M67 平台数据导入回灌 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 平台导出文件（xlsx/csv）→ 服务端解析 → 通用列映射 → 自动匹配回灌 `publications` 表，补全「发布 → 平台数据 → 复盘」闭环唯一缺失的数据入口。

**Architecture:** 无状态单端点两阶段（`dry_run=1` 预览 / `dry_run=0` 落库，文件传两次）；解析与匹配全纯函数（`services/table-parse.ts` + `services/publication-import.ts`），路由仅做 IO 装配；前端在发布面板加三步导入弹层。复盘消费方 `publication_ingest` 零改动直接受益。

**Tech Stack:** Node 22+/TypeScript、Hono（既有路由）、fflate（既有依赖，xlsx 解包）、Drizzle + libsql（既有）、Vue 3 `<script setup>`（前端）。

**Spec:** `docs/records/M67-平台数据导入回灌-design.md`（计划从 spec 论证，执行者须同读）。

## Global Constraints

- 红线：**0 新表 0 新列 0 新依赖 0 新付费面**（fflate 已有 + Node 内置 ICU/TextDecoder）。
- 现有 `/publications*` 端点与 `publication_ingest` 行为**零变更**（纯加法）。
- `dry_run=1` 两种形态（含/无 mapping）**零写库**（探针断言）。
- 解析与匹配全纯函数、零 IO；路由只做 form 解析 / db / 装配。
- 上限：文件 ≤10MB（`TABLE_MAX_FILE_BYTES`）、数据行 ≤2000 不含表头（`TABLE_MAX_ROWS`），超限抛错。
- `platform` 枚举复用 `routes/publications.ts` 既有 `PLATFORMS` 常量（`douyin|wechat_channels|kuaishou|xiaohongshu|bilibili|other`）。
- probe 断言文案不得含 `PASS`/`FAIL` 词元（`parseProbeOutput` 会误计）。
- fflate 字符串互转：`strFromU8`/`strToU8` 一律**无第二参**（UTF-8 缺省；第二参 true = latin1，见 `doc-parse.ts:13` 先例注释）。
- 提交：conventional commits 中文，前缀 `feat(m67)/fix(m67)/test(m67)/docs(m67)`；**禁 `git add -A`**，逐文件 add。
- 模板 YAML 零改动；不进 DAG（面板能力，非流水线步骤）。
- probe 命令：`cd apps/server && npx tsx scripts/probe-m67.ts [--section=parse|match|route]`；typecheck：`pnpm --filter @acs/server typecheck`（server）/ `pnpm --filter @acs/web typecheck`（web，以 package.json name 实际为准）。

```ts
/**
 * 表格解析（M67 平台数据导入回灌，spec §3.1）：xlsx / csv → 二维字符串表，纯函数零 IO。
 * 0 新依赖：xlsx 走 fflate 解包 + 最小 sheet XML 读取器（与 doc-parse.ts 同风格）；
 * CSV 手写 RFC4180；GBK 编码走 Node 内置 ICU TextDecoder('gb18030') 兜底。
 * 边界（spec §五）：合并单元格不展开、多 sheet 只读第一张、BIFF 二进制 xls 不支持。
 */
import { unzipSync, strFromU8 } from 'fflate'

export type TableParseCode =
  | 'too_large'
  | 'too_many_rows'
  | 'bad_ext'
  | 'bad_zip'
  | 'old_xls'
  | 'no_sheet'

export class TableParseError extends Error {
  constructor(
    readonly code: TableParseCode,
    message: string,
  ) {
    super(message)
    this.name = 'TableParseError'
  }
}

export interface ParsedTable {
  header: string[]
  rows: string[][]
  warnings: string[]
}

export const TABLE_MAX_FILE_BYTES = 10 * 1024 * 1024
export const TABLE_MAX_ROWS = 2000

// ⚠ fflate 第二参语义：true = latin1（中文乱码），缺省 = UTF-8（doc-parse.ts:13 先例）
const utf8 = (b: Uint8Array): string => strFromU8(b)

function unescapeXml(s: string): string {
  return s
    .replace(/&#x([0-9a-fA-F]+);/g, (_, h: string) => String.fromCodePoint(parseInt(h, 16)))
    .replace(/&#(\d+);/g, (_, d: string) => String.fromCodePoint(Number(d)))
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&apos;/g, "'")
    .replace(/&amp;/g, '&')
}

/** Excel 1900 序列号 → ms（epoch 1899-12-30；浮点四舍五入到毫秒） */
function serialToMs(serial: number): number {
  return Math.round((serial - 25569) * 86_400_000)
}

/** 序列号按无时区朴素时间格式化（UTC 字段取值 → 与 Excel 所见一致；下游按本地解析，端到端语义一致） */
function fmtNaive(ms: number): string {
  const d = new Date(ms)
  const p = (n: number): string => String(n).padStart(2, '0')
  return `${d.getUTCFullYear()}-${p(d.getUTCMonth() + 1)}-${p(d.getUTCDate())} ${p(d.getUTCHours())}:${p(d.getUTCMinutes())}`
}

/** styles.xml → 样式索引是否日期格式（内置 14–22 或自定义 formatCode 含 y/m/d/h） */
function loadDateStyles(stylesXml: string): (s: number) => boolean {
  const custom = new Map<string, string>()
  for (const m of stylesXml.matchAll(/<numFmt\b[^>]*numFmtId="(\d+)"[^>]*formatCode="([^"]*)"[^>]*\/?>/g)) {
    custom.set(m[1]!, m[2]!)
  }
  const xfsBlock = /<cellXfs\b[^>]*>([\s\S]*?)<\/cellXfs>/.exec(stylesXml)
  const fmtIds: string[] = []
  if (xfsBlock) for (const m of xfsBlock[1]!.matchAll(/<xf\b[^>]*numFmtId="(\d+)"[^>]*\/?>/g)) fmtIds.push(m[1]!)
  const isDateFmt = (id: string): boolean => {
    const n = Number(id)
    if (n >= 14 && n <= 22) return true
    const code = custom.get(id)
    return !!code && /[ymdh]/i.test(code)
  }
  return (s: number) => s >= 0 && s < fmtIds.length && isDateFmt(fmtIds[s]!)
}

/** A1 引用 → 0 基列索引（A=0） */
function colIndex(ref: string): number {
  let n = 0
  for (const ch of ref) {
    if (ch >= 'A' && ch <= 'Z') n = n * 26 + (ch.charCodeAt(0) - 64)
    else break
  }
  return n - 1
}

function parseXlsx(buf: Buffer): ParsedTable {
  let files: Record<string, Uint8Array>
  try {
    files = unzipSync(new Uint8Array(buf))
  } catch {
    throw new TableParseError('bad_zip', 'xlsx 解包失败（加密或损坏？）')
  }
  const sheetKey = Object.keys(files).find((k) => /^xl\/worksheets\/sheet1\.xml$/i.test(k))
  if (!sheetKey) throw new TableParseError('no_sheet', 'xlsx 缺 xl/worksheets/sheet1.xml（多 sheet 仅读第一张）')
  const sheetXml = utf8(files[sheetKey]!)

  const warnings: string[] = []
  const shared: string[] = []
  const sstKey = Object.keys(files).find((k) => /^xl\/sharedStrings\.xml$/i.test(k))
  if (sstKey) {
    for (const m of utf8(files[sstKey]!).matchAll(/<si\b[^>]*>([\s\S]*?)<\/si>/g)) {
      let text = ''
      for (const t of m[1]!.matchAll(/<t\b[^>]*(?:\/>|>([\s\S]*?)<\/t>)/g)) text += t[1] ?? ''
      shared.push(unescapeXml(text))
    }
  }

  let isDateStyle: ((s: number) => boolean) | null = null
  const stylesKey = Object.keys(files).find((k) => /^xl\/styles\.xml$/i.test(k))
  if (stylesKey) isDateStyle = loadDateStyles(utf8(files[stylesKey]!))

  // 稀疏网格：行号/列号定位，空缺补空串
  const grid = new Map<number, Map<number, string>>()
  for (const rowM of sheetXml.matchAll(/<row\b[^>]*\br="(\d+)"[^>]*>([\s\S]*?)<\/row>/g)) {
    const r = Number(rowM[1]!)
    for (const cM of rowM[2]!.matchAll(/<c\b([^>]*?)(?:\/>|>([\s\S]*?)<\/c>)/g)) {
      const attrs = cM[1]!
      const ref = /\br="([A-Z]+)\d+"/.exec(attrs)?.[1] ?? ''
      const col = ref ? colIndex(ref) : -1
      if (col < 0) continue
      const t = /\bt="([^"]+)"/.exec(attrs)?.[1] ?? 'n'
      const sAttr = /\bs="(\d+)"/.exec(attrs)?.[1]
      const inner = cM[2] ?? ''
      let val: string | null = null
      if (t === 'inlineStr') {
        let text = ''
        for (const tm of inner.matchAll(/<t\b[^>]*>([\s\S]*?)<\/t>/g)) text += tm[1]!
        val = unescapeXml(text)
      } else {
        const v = /<v>([\s\S]*?)<\/v>/.exec(inner)?.[1]
        if (v !== undefined) {
          if (t === 's') {
            const idx = Number(v)
            if (Number.isInteger(idx) && idx >= 0 && idx < shared.length) val = shared[idx]!
            else warnings.push(`共享字符串索引越界：${v}`)
          } else {
            val = unescapeXml(v)
            // t="str"（公式缓存文本）与 t="b" 均按字符串处理；日期判定仅对数字形态生效
            if (t !== 'str' && isDateStyle?.(Number(sAttr ?? -1))) {
              const n = Number(val)
              if (Number.isFinite(n) && n > 0) val = fmtNaive(serialToMs(n))
            }
          }
        }
      }
      if (val !== null) {
        let row = grid.get(r)
        if (!row) grid.set(r, (row = new Map()))
        row.set(col, val)
      }
    }
  }

  const rowKeys = [...grid.keys()].sort((a, b) => a - b)
  while (rowKeys.length > 0) {
    const last = rowKeys[rowKeys.length - 1]!
    const cells = grid.get(last)!
    if (cells.size === 0 || [...cells.values()].every((c) => c === '')) rowKeys.pop()
    else break
  }
  if (rowKeys.length === 0) return { header: [], rows: [], warnings }
  if (rowKeys.length - 1 > TABLE_MAX_ROWS) {
    throw new TableParseError('too_many_rows', `数据行 ${rowKeys.length - 1} 超上限 ${TABLE_MAX_ROWS}`)
  }
  const width = Math.max(...rowKeys.map((r) => Math.max(...grid.get(r)!.keys())) + 1)
  const dense = rowKeys.map((r) => {
    const cells = grid.get(r)!
    const row: string[] = []
    for (let j = 0; j < width; j++) row.push(cells.get(j) ?? '')
    return row
  })
  const [header, ...rows] = dense
  return { header: header ?? [], rows, warnings }
}

function parseCsv(buf: Buffer): ParsedTable {
  let text: string
  try {
    text = new TextDecoder('utf-8', { fatal: true }).decode(buf)
  } catch {
    text = new TextDecoder('gb18030').decode(buf)
  }
  if (text.charCodeAt(0) === 0xfeff) text = text.slice(1)

  const rows: string[][] = []
  let row: string[] = []
  let field = ''
  let inQuotes = false
  let touched = false
  const pushField = (): void => {
    row.push(field)
    field = ''
  }
  const pushRow = (): void => {
    pushField()
    if (touched || row.some((c) => c !== '')) rows.push(row)
    row = []
    touched = false
  }
  for (let i = 0; i < text.length; i++) {
    const ch = text[i]!
    if (inQuotes) {
      if (ch === '"') {
        if (text[i + 1] === '"') {
          field += '"'
          i++
        } else inQuotes = false
      } else field += ch
    } else if (ch === '"') {
      inQuotes = true
      touched = true
    } else if (ch === ',') {
      pushField()
      touched = true
    } else if (ch === '\n' || ch === '\r') {
      if (ch === '\r' && text[i + 1] === '\n') i++
      pushRow()
    } else {
      field += ch
      touched = true
    }
  }
  if (field !== '' || row.length > 0 || touched) pushRow()

  if (rows.length - 1 > TABLE_MAX_ROWS) {
    throw new TableParseError('too_many_rows', `数据行 ${rows.length - 1} 超上限 ${TABLE_MAX_ROWS}`)
  }
  const [header, ...data] = rows
  return { header: header ?? [], rows: data, warnings: [] }
}

/** 入口：按扩展名分发；.xls 先嗅探 PK 魔数（不少平台导出实为 xlsx 改名） */
export function parseTable(buf: Buffer, ext: string): ParsedTable {
  if (buf.byteLength > TABLE_MAX_FILE_BYTES) {
    throw new TableParseError('too_large', `文件 ${buf.byteLength} 字节超上限 ${TABLE_MAX_FILE_BYTES}`)
  }
  const e = ext.toLowerCase().replace(/^\./, '')
  if (e === 'csv') return parseCsv(buf)
  if (e === 'xlsx' || e === 'xls') {
    if (buf.subarray(0, 2).toString('latin1') !== 'PK') {
      throw new TableParseError('old_xls', '旧版二进制 xls 不支持，请另存为 xlsx/csv')
    }
    return parseXlsx(buf)
  }
  throw new TableParseError('bad_ext', `不支持的扩展名：${ext}（支持 xlsx/csv）`)
}
```

```ts
/**
 * 发布数据导入装配与匹配（M67，spec §3.2）：纯函数，零 IO、零 db。
 * 更新保护靠「patch 白名单」构造性成立：patch 只含 metrics/url/publishedAt/title 四键，
 * runId/assetId/abGroup/note 永不出现在更新集——路由只 set patch，保护无需额外判断。
 */
import type { Publication } from '../db/schema'
import type { ParsedTable } from './table-parse'

export type MappingField =
  | 'title' | 'url' | 'publishedAt'
  | 'views' | 'likes' | 'comments' | 'favorites' | 'shares'
  | 'abGroup'

export type ImportMapping = Partial<Record<MappingField, number>>

export const MAPPING_FIELDS: readonly MappingField[] = [
  'title', 'url', 'publishedAt', 'views', 'likes', 'comments', 'favorites', 'shares', 'abGroup',
]

export class ImportError extends Error {
  constructor(
    readonly code: string,
    message: string,
  ) {
    super(message)
    this.name = 'ImportError'
  }
}

export interface AssembledRow {
  rowIndex: number
  platform: string
  title: string | null
  url: string | null
  publishedAt: number | null
  abGroup: string | null
  metrics: { views: number; likes: number; comments: number; favorites: number; shares: number }
}

export interface ImportPlan {
  updates: Array<{
    id: number
    row: AssembledRow
    patch: { metrics: string; url: string | null; publishedAt: number | null; title: string | null }
  }>
  creates: AssembledRow[]
  duplicates: number
}

const METRIC_KEYS: readonly (keyof AssembledRow['metrics'])[] = ['views', 'likes', 'comments', 'favorites', 'shares']

export function parseMapping(raw: unknown): ImportMapping {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
    throw new ImportError('bad_mapping', 'mapping 需为对象')
  }
  const out: ImportMapping = {}
  let useful = 0
  for (const [k, v] of Object.entries(raw as Record<string, unknown>)) {
    if (!(MAPPING_FIELDS as readonly string[]).includes(k)) {
      throw new ImportError('bad_mapping', `mapping 未知字段：${k}`)
    }
    if (v === null || v === undefined || v === '') continue
    if (!Number.isInteger(v) || (v as number) < 0) {
      throw new ImportError('bad_mapping', `mapping.${k} 需为非负整数列索引`)
    }
    out[k as MappingField] = v as number
    useful++
  }
  if (useful === 0) throw new ImportError('bad_mapping', 'mapping 至少映射一个字段')
  return out
}

const DATE_RE = /^(\d{4})[-/](\d{1,2})[-/](\d{1,2})(?:[ T](\d{1,2}):(\d{2})(?::(\d{2}))?)?$/

/** 日期字符串 → ms：xlsx 路产出的 'YYYY-MM-DD HH:mm' 与 CSV 文本（2026/1/5 10:00、2026-01-05）按本地时区解析 */
export function toMs(s: string | null): number | null {
  if (!s) return null
  const t = s.trim()
  const m = DATE_RE.exec(t)
  if (m) {
    return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]), Number(m[4] ?? 0), Number(m[5] ?? 0), Number(m[6] ?? 0)).getTime()
  }
  if (/^\d{12,13}$/.test(t)) return Number(t)
  return null
}

function dayOf(ms: number | null): string {
  if (ms === null) return ''
  const d = new Date(ms)
  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`
}

function cellOf(row: string[], idx: number): string | null {
  return idx >= 0 && idx < row.length ? row[idx]!.trim() : null
}

function numOf(s: string | null): number {
  if (!s) return 0
  const n = Number(s.replace(/[,，\s]/g, ''))
  return Number.isFinite(n) && n > 0 ? Math.floor(n) : 0
}

export function assembleRows(
  table: ParsedTable,
  mapping: ImportMapping,
  platform: string,
): { rows: AssembledRow[]; warnings: string[] } {
  const warnings: string[] = []
  const warnedCols = new Set<MappingField>()
  const pick = (r: string[], f: MappingField): string | null => {
    const idx = mapping[f]
    if (idx === undefined) return null
    if (idx >= r.length) {
      if (!warnedCols.has(f)) {
        warnings.push(`列索引越界：${f}（第 ${idx} 列不存在，按缺失处理）`)
        warnedCols.add(f)
      }
      return null
    }
    return cellOf(r, idx)
  }
  const rows: AssembledRow[] = table.rows.map((r, i) => {
    const rawAt = pick(r, 'publishedAt')
    const ms = toMs(rawAt)
    if (rawAt && ms === null) warnings.push(`第 ${i + 2} 行发布时间无法解析：${rawAt.slice(0, 40)}（置空）`)
    return {
      rowIndex: i,
      platform,
      title: pick(r, 'title') || null,
      url: pick(r, 'url') || null,
      publishedAt: ms,
      abGroup: pick(r, 'abGroup') || null,
      metrics: {
        views: numOf(pick(r, 'views')),
        likes: numOf(pick(r, 'likes')),
        comments: numOf(pick(r, 'comments')),
        favorites: numOf(pick(r, 'favorites')),
        shares: numOf(pick(r, 'shares')),
      },
    }
  })
  return { rows, warnings }
}

type Metrics = AssembledRow['metrics']

function safeMetrics(raw: string | null): Metrics {
  const out: Metrics = { views: 0, likes: 0, comments: 0, favorites: 0, shares: 0 }
  try {
    const m = JSON.parse(raw ?? '{}') as Record<string, unknown>
    for (const k of METRIC_KEYS) {
      const n = Number(m[k] ?? 0)
      if (Number.isFinite(n) && n > 0) out[k] = Math.floor(n)
    }
  } catch {
    /* 损坏 JSON 按全零 */
  }
  return out
}

/** 文件内去重键：URL 优先；无 URL 用 平台|标题|同日 */
function dedupeKey(r: AssembledRow): string {
  if (r.url) return `u:${r.url}`
  return `t:${r.platform}|${r.title ?? ''}|${dayOf(r.publishedAt)}`
}

export function matchRows(rows: AssembledRow[], existing: Publication[]): ImportPlan {
  const plan: ImportPlan = { updates: [], creates: [], duplicates: 0 }
  // 同文件去重：同匹配键取最后一条（后行覆盖前行）
  const lastByKey = new Map<string, AssembledRow>()
  for (const r of rows) lastByKey.set(dedupeKey(r), r)
  plan.duplicates = rows.length - lastByKey.size

  const claimed = new Set<number>()
  for (const r of lastByKey.values()) {
    let hit: Publication | undefined
    if (r.url) {
      hit = existing.find((e) => e.platform === r.platform && (e.url ?? '').trim() === r.url)
    }
    if (!hit && r.title) {
      hit = existing.find(
        (e) =>
          e.platform === r.platform &&
          (e.title ?? '').trim() === r.title &&
          e.publishedAt !== null &&
          r.publishedAt !== null &&
          dayOf(e.publishedAt) === dayOf(r.publishedAt),
      )
    }
    if (hit && !claimed.has(hit.id)) {
      claimed.add(hit.id)
      const ex = safeMetrics(hit.metrics)
      const merged = {} as Metrics
      for (const k of METRIC_KEYS) merged[k] = r.metrics[k] > 0 ? r.metrics[k] : ex[k]
      const exUrl = (hit.url ?? '').trim()
      const exTitle = (hit.title ?? '').trim()
      plan.updates.push({
        id: hit.id,
        row: r,
        patch: {
          metrics: JSON.stringify(merged),
          url: exUrl ? exUrl : r.url,
          publishedAt: hit.publishedAt ?? r.publishedAt,
          title: exTitle ? exTitle : r.title,
        },
      })
    } else {
      plan.creates.push(r)
    }
  }
  return plan
}
```
