# M69 审片问题单与返修复检 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把「看片发现的问题」落成 `review_findings` 台账：可定位（镜头/时间窗/全片）、可派给既有字幕与合成输入返修、可在新成片上做证据复检，并由人终判 `verified`。

**Architecture:** 1 张新表 + 4 个后端模块（`scope` 时间轴定位纯函数、`snapshot` 证据快照、`recheck` 证据差异、`service` 状态机）+ 1 个只读/写入路由 + 1 个运行详情右栏面板。时间轴唯一真源仍是成片 `params.timeline`；复检只报 `changed|unchanged|unverifiable`，永不自动判过；不改 QC 10 键、不改批准链、不阻断导出。

**Tech Stack:** Hono + Drizzle/libsql（SQLite）、Vue 3 + Vite、tsx 探针（`isolatedEnv`/`makeChecker`/`runSections`）。

**Spec:** `docs/records/M69-审片问题单与返修复检-design.md`

## Global Constraints

- 唯一新增持久化 = `review_findings` 1 张表；既有表 0 新列。
- 0 新依赖、0 新付费面、0 模型调用；不触模板 YAML、QC 10 键、批准链、交付标记。
- 迁移只由 `pnpm --filter @acs/server db:generate` 生成 `apps/server/drizzle/0002_*.sql`，不手写/手改。
- 单文件 ≤ 800 行；前端沿用全局 `.panel.mini` / `.badge` / `.btn` 主题 token，不新造配色。
- 所有时间毫秒整数；资产 `assets.duration` 单位为**秒**，转毫秒需 `* 1000`。
- 复检结果恒 `manual_required: true`（本表不存该字段，语义由 `recheck_state` + 文案保证）；`changed` ≠ 修好，`unchanged` ≠ 没修好。
- 失败请求零写库；领域错误统一 `error.code` 机读（404/400/409）。
- 探针零网络、零付费；不使用真实用户媒体。

```ts
/**
 * M69 审片问题单 · 时间轴定位纯函数（design §4.3）。
 * 唯一真源 = 成片 params.timeline：segments.startSec 为内容轴（片头后 0 起算），
 * cues.coordinate='final' 已是最终绝对轴。本模块不查库、不读文件、不触执行。
 */
export type ReviewScopeKind = 'shot' | 'time' | 'global'
export type ReviewCategory = 'visual' | 'prop' | 'lip_sync' | 'subtitle' | 'audio' | 'pacing' | 'other'
export type ReviewSeverity = 'blocker' | 'major' | 'minor'

export interface TimelineSegment {
  shotId: string | null
  assetId: number | null
  kind: string
  startMs: number
  endMs: number
  durMs: number
}

export interface TimelineCue {
  id: string
  startMs: number
  endMs: number
  text: string
}

export interface FinalTimeline {
  totalMs: number
  introMs: number
  outroMs: number
  segments: TimelineSegment[]
  cues: TimelineCue[]
}

export interface ReviewAssetLike {
  id: number
  kind: string
  purpose?: string | null
  relPath?: string | null
  deletedAt?: number | null
  sha256?: string | null
  duration?: number | null
  createdAt?: number | null
}

export interface ReviewStepLike {
  id: number
  stepKey: string
  actionKey: string
  output: string | null
}

export interface ResolvedFinal {
  assetId: number
  /** 是否走了「该步无可用产物 → 全 run 最新成片」降级 */
  degraded: boolean
}

const isObj = (v: unknown): v is Record<string, unknown> =>
  !!v && typeof v === 'object' && !Array.isArray(v)

const num = (v: unknown): number | null =>
  typeof v === 'number' && Number.isFinite(v) ? v : null

function outputIdsOfRaw(output: string | null): number[] {
  if (!output) return []
  try {
    const o = JSON.parse(output) as Record<string, unknown>
    const ids = o['asset_ids']
    if (!Array.isArray(ids)) return []
    return ids.map(Number).filter((n) => Number.isInteger(n) && n > 0)
  } catch {
    return []
  }
}

function usableFinal(a: ReviewAssetLike): boolean {
  return (
    a.deletedAt == null &&
    a.kind === 'video' &&
    a.purpose === 'final_video' &&
    typeof a.relPath === 'string' &&
    a.relPath.length > 0
  )
}

/**
 * 该 stepKey 的当前成片：先取同 stepKey + ffmpeg_merge 步骤 output.asset_ids 中最新可用 final_video；
 * 无匹配步骤或产物被删 → 回退该 run 最新可用 final_video（降级，仍可审）。
 */
export function resolveReviewFinal(
  steps: ReviewStepLike[],
  assets: ReviewAssetLike[],
  stepKey = 'compose',
): ResolvedFinal | null {
  const byId = new Map(assets.map((a) => [a.id, a]))
  const compose = steps
    .filter((s) => s.stepKey === stepKey && s.actionKey === 'ffmpeg_merge')
    .sort((a, b) => b.id - a.id)[0]
  if (compose) {
    const ids = outputIdsOfRaw(compose.output)
    for (const id of [...ids].sort((a, b) => b - a)) {
      const a = byId.get(id)
      if (a && usableFinal(a)) return { assetId: a.id, degraded: false }
    }
  }
  const fallback = assets
    .filter(usableFinal)
    .sort((a, b) => b.id - a.id)[0]
  return fallback ? { assetId: fallback.id, degraded: true } : null
}

/** 只读 params.timeline；结构不足返回 null（降级，不反推） */
export function parseFinalTimeline(paramsJson: string | null): FinalTimeline | null {
  if (!paramsJson) return null
  let params: Record<string, unknown>
  try {
    const p = JSON.parse(paramsJson) as unknown
    if (!isObj(p)) return null
    params = p
  } catch {
    return null
  }
  const raw = params['timeline']
  if (!isObj(raw)) return null
  const introMs = Math.max(0, Math.round((num(raw['introSec']) ?? 0) * 1000))
  const outroMs = Math.max(0, Math.round((num(raw['outroSec']) ?? 0) * 1000))

  const rawSegs = raw['segments']
  const segments: TimelineSegment[] = Array.isArray(rawSegs)
    ? rawSegs.filter(isObj).map((s) => {
        const startSec = num(s['startSec']) ?? 0
        const durSec = Math.max(0, num(s['durSec']) ?? 0)
        const startMs = introMs + Math.round(startSec * 1000)
        return {
          shotId: typeof s['shotId'] === 'string' && s['shotId'] ? s['shotId'] : null,
          assetId: num(s['assetId']),
          kind: typeof s['kind'] === 'string' ? s['kind'] : 'video',
          startMs,
          endMs: startMs + Math.round(durSec * 1000),
          durMs: Math.round(durSec * 1000),
        }
      })
    : []

  const totalSec = num(raw['totalSec'])
  const derived = introMs + segments.reduce((n, s) => n + s.durMs, 0) + outroMs
  const totalMs = totalSec !== null ? Math.round(totalSec * 1000) : derived
  // 既无声明总长、又无任何段 → 无时间轴证据
  if (totalSec === null && segments.length === 0) return null

  const sub = raw['subtitle']
  const rawCues = isObj(sub) ? sub['cues'] : undefined
  const cues: TimelineCue[] = Array.isArray(rawCues)
    ? rawCues
        .filter(isObj)
        .map((c) => ({
          id: typeof c['id'] === 'string' ? c['id'] : '',
          startMs: Math.round(num(c['startMs']) ?? 0),
          endMs: Math.round(num(c['endMs']) ?? 0),
          text: typeof c['text'] === 'string' ? c['text'] : '',
        }))
        .filter((c) => c.endMs > c.startMs)
    : []

  return { totalMs, introMs, outroMs, segments, cues }
}

export interface AtLocation {
  kind: 'intro' | 'shot' | 'outro' | 'outside'
  shotId?: string
  startMs?: number
  endMs?: number
}

/** 半开区间定位：边界取后一段；atMs ≥ totalMs 或负值 → outside */
export function locateAt(tl: FinalTimeline, atMs: number): AtLocation {
  if (!Number.isFinite(atMs) || atMs < 0 || atMs >= tl.totalMs) return { kind: 'outside' }
  if (atMs < tl.introMs) return { kind: 'intro' }
  for (const s of tl.segments) {
    if (atMs >= s.startMs && atMs < s.endMs) {
      return { kind: 'shot', shotId: s.shotId ?? undefined, startMs: s.startMs, endMs: s.endMs }
    }
  }
  return { kind: 'outro' }
}

export function shotRange(tl: FinalTimeline, shotId: string): { startMs: number; endMs: number } | null {
  const s = tl.segments.find((x) => x.shotId === shotId)
  return s ? { startMs: s.startMs, endMs: s.endMs } : null
}

export type TimeRangeCheck =
  | { ok: true }
  | { ok: false; code: 'bad_scope' | 'no_timeline'; message: string }

/** 时间窗：整数、0 ≤ start < end、end ≤ 上限（timeline.totalMs，缺省资产时长毫秒） */
export function validateTimeRange(
  tl: FinalTimeline | null,
  startMs: number,
  endMs: number,
  fallbackDurationMs: number | null,
): TimeRangeCheck {
  if (!Number.isInteger(startMs) || !Number.isInteger(endMs)) {
    return { ok: false, code: 'bad_scope', message: '时间范围需为整数毫秒' }
  }
  if (startMs < 0 || startMs >= endMs) {
    return { ok: false, code: 'bad_scope', message: '时间范围需满足 0 ≤ 起点 < 终点' }
  }
  const max = tl?.totalMs ?? fallbackDurationMs
  if (max === null) {
    return { ok: false, code: 'no_timeline', message: '当前成片缺少可用时间轴，无法创建时间范围问题；请改用全片范围' }
  }
  if (endMs > max) {
    return { ok: false, code: 'bad_scope', message: `时间范围超出成片时长（${max}ms）` }
  }
  return { ok: true }
}

/** 4.4 范围/类别矩阵：lip_sync 仅 shot；pacing 仅 time/global；其余不限 */
export function categoryAllowsScope(category: ReviewCategory, kind: ReviewScopeKind): boolean {
  if (category === 'lip_sync') return kind === 'shot'
  if (category === 'pacing') return kind === 'time' || kind === 'global'
  return true
}
```

```ts
/**
 * M69 · 基准证据快照（design §4.2/§4.6）。
 * 只存「创建问题时看到的证据」，只覆盖问题范围：shot → 该镜段 + 相交 cues；
 * time → 相交段 + 窗口 cues；global → 全片。cues 50 / segments 200 上限，超限只留摘要。
 * 纯函数：sha256 由调用方注入（服务层 join assets）。

import { createHash } from 'node:crypto'
import { shotRange, type FinalTimeline, type TimelineCue, type TimelineSegment } from './scope'

export interface SnapshotMedia {
  shotId: string | null
  assetId: number | null
  sha256: string | null
  startMs: number
  endMs: number
  durMs: number
}

export interface SnapshotShot extends SnapshotMedia {
  shotId: string
}

export interface SnapshotCue {
  id: string
  startMs: number
  endMs: number
  text: string
}

export interface SnapshotAudio {
  bgm: { assetId: number | null; sha256: string | null; volume: number | null; fadeSec: number | null } | null
  sfx: Array<{ shotId: string | null; assetId: number | null; sha256: string | null }>

  v: 1
  stepKey: string
  finalAssetId: number
  finalSha256: string | null
  totalMs: number
  introMs: number
  shot: SnapshotShot | null
  segments: SnapshotMedia[]
  segmentsTruncated: boolean
  cues: SnapshotCue[]
  cuesDigest: string | null
  cuesCount: number
  cuesTruncated: boolean
  audio: SnapshotAudio | null

export type SnapshotScope =
  | { kind: 'shot'; shotId: string }
  | { kind: 'time'; startMs: number; endMs: number }
  | { kind: 'global' }

export interface TimelineAudioInput {
  bgm: { assetId: number | null; volume: number | null; fadeSec: number | null } | null
  sfx: Array<{ shotId: string | null; assetId: number | null }>
}

export type AssetShaLookup = (assetId: number | null) => string | null

const isObj = (v: unknown): v is Record<string, unknown> =>
  !!v && typeof v === 'object' && !Array.isArray(v)

const num = (v: unknown): number | null =>
  typeof v === 'number' && Number.isFinite(v) ? v : null

const str = (v: unknown): string | null => (typeof v === 'string' ? v : null)

export const CUE_LIMIT = 50
export const SEGMENT_LIMIT = 200

/** 读 params.timeline 的音频实际生效值（不是 run.input._compose 覆盖意图）；无 timeline → null */
export function parseTimelineAudio(paramsJson: string | null): TimelineAudioInput | null {
  if (!paramsJson) return null
  let params: Record<string, unknown>
  try {
    const p = JSON.parse(paramsJson) as unknown
    if (!isObj(p)) return null
    params = p
  } catch {
    return null
  }
  const tl = params['timeline']
  if (!isObj(tl)) return null
  const rawBgm = tl['bgm']
  const bgm = isObj(rawBgm)
    ? { assetId: num(rawBgm['assetId']), volume: num(rawBgm['volume']), fadeSec: num(rawBgm['fadeSec']) }
    : null
  const rawSfx = tl['sfx']
  const sfx = Array.isArray(rawSfx)
    ? rawSfx.filter(isObj).map((s) => ({
        shotId: str(s['shotId']),
        assetId: num(s['assetId']),
      }))
    : []
  return { bgm, sfx }
}

/** 规范化 cue 摘要：排序后 [startMs,endMs,text]，不含 id（id 可能随重排变化） */
export function cuesDigestOf(cues: SnapshotCue[]): string {
  const norm = [...cues]
    .sort((a, b) => a.startMs - b.startMs || a.endMs - b.endMs || a.text.localeCompare(b.text))
    .map((c) => [c.startMs, c.endMs, c.text])
  return createHash('sha256').update(JSON.stringify(norm)).digest('hex')
}

function toMedia(s: TimelineSegment, assetSha: AssetShaLookup): SnapshotMedia {
  return {
    shotId: s.shotId,
    assetId: s.assetId,
    sha256: assetSha(s.assetId),
    startMs: s.startMs,
    endMs: s.endMs,
    durMs: s.durMs,
  }
}

export function buildBaseSnapshot(p: {
  stepKey: string
  finalAssetId: number
  finalSha256: string | null
  /** 无时间轴的旧成片允许为 null：只记录「无证据」快照，复检将 unverifiable */
  timeline: FinalTimeline | null
  scope: SnapshotScope
  audio: TimelineAudioInput | null
  assetSha: AssetShaLookup
  limits?: { cues?: number; segments?: number }
}): BaseSnapshot {
  const { timeline, scope, assetSha } = p
  const cueLimit = p.limits?.cues ?? CUE_LIMIT
  const segLimit = p.limits?.segments ?? SEGMENT_LIMIT

  const win: { startMs: number; endMs: number } | null = timeline
    ? scope.kind === 'shot'
      ? shotRange(timeline, scope.shotId)
      : scope.kind === 'time'
        ? { startMs: scope.startMs, endMs: scope.endMs }
        : { startMs: 0, endMs: timeline.totalMs }
    : scope.kind === 'time'
      ? { startMs: scope.startMs, endMs: scope.endMs }
      : null

  const shotSeg =
    timeline && scope.kind === 'shot' && win
      ? timeline.segments.find((s) => s.shotId === scope.shotId) ?? null
      : null
  const shot: SnapshotShot | null =
    shotSeg && shotSeg.shotId
      ? { ...toMedia(shotSeg, assetSha), shotId: shotSeg.shotId }
      : null

  const allSegs = timeline && win ? timeline.segments.filter((s) => s.endMs > win.startMs && s.startMs < win.endMs) : []
  const segments = allSegs.slice(0, segLimit).map((s) => toMedia(s, assetSha))

  const allCues: SnapshotCue[] =
    timeline && win
      ? timeline.cues
          .filter((c) => c.endMs > win.startMs && c.startMs < win.endMs)
          .map((c: TimelineCue) => ({ id: c.id, startMs: c.startMs, endMs: c.endMs, text: c.text }))
      : []

  let audio: SnapshotAudio | null = null
  if (p.audio) {
    const sfx = scope.kind === 'shot'
      ? p.audio.sfx.filter((s) => s.shotId === (scope as { shotId: string }).shotId)
      : p.audio.sfx
    audio = {
      bgm: p.audio.bgm
        ? {
            assetId: p.audio.bgm.assetId,
            sha256: assetSha(p.audio.bgm.assetId),
            volume: p.audio.bgm.volume,
            fadeSec: p.audio.bgm.fadeSec,
          }
        : null,
      sfx: sfx.map((s) => ({ shotId: s.shotId, assetId: s.assetId, sha256: assetSha(s.assetId) })),
    }
  }

  return {
    v: 1,
    stepKey: p.stepKey,
    finalAssetId: p.finalAssetId,
    finalSha256: p.finalSha256,
    totalMs: timeline?.totalMs ?? 0,
    introMs: timeline?.introMs ?? 0,
    shot,
    segments,
    segmentsTruncated: allSegs.length > segLimit,
    cues: allCues.slice(0, cueLimit),
    cuesDigest: cuesDigestOf(allCues),
    cuesCount: allCues.length,
    cuesTruncated: allCues.length > cueLimit,
    audio,
  }
}

const isMedia = (v: unknown): v is SnapshotMedia => isObj(v)
const isCue = (v: unknown): v is SnapshotCue => isObj(v)

/** 读回快照；结构不足 → null（复检降级 unverifiable） */
export function parseBaseSnapshot(raw: string | null): BaseSnapshot | null {
  if (!raw) return null
  let o: Record<string, unknown>
  try {
    const p = JSON.parse(raw) as unknown
    if (!isObj(p)) return null
    o = p
  } catch {
    return null
  }
  if (o['v'] !== 1 || num(o['finalAssetId']) === null) return null
  const shotRaw = o['shot']
  return {
    v: 1,
    stepKey: str(o['stepKey']) ?? 'compose',
    finalAssetId: num(o['finalAssetId'])!,
    finalSha256: str(o['finalSha256']),
    totalMs: num(o['totalMs']) ?? 0,
    introMs: num(o['introMs']) ?? 0,
    shot: isObj(shotRaw) && typeof shotRaw['shotId'] === 'string'
      ? (shotRaw as unknown as SnapshotShot)
      : null,
    segments: Array.isArray(o['segments']) ? (o['segments'] as unknown[]).filter(isMedia) : [],
    segmentsTruncated: o['segmentsTruncated'] === true,
    cues: Array.isArray(o['cues']) ? (o['cues'] as unknown[]).filter(isCue) : [],
    cuesDigest: str(o['cuesDigest']),
    cuesCount: num(o['cuesCount']) ?? 0,
    cuesTruncated: o['cuesTruncated'] === true,
    audio: isObj(o['audio']) ? (o['audio'] as unknown as SnapshotAudio) : null,
  }
}
```

```ts
/**
 * M69 · 复检（design §4.6）：只报证据变化，永不判过。
 * 比较基准快照与当前 step_key 成片时间轴；任何结果都由人确认后才 verified。
 */
import { shotRange, type FinalTimeline, type ReviewCategory, type ReviewScopeKind } from './scope'
import { cuesDigestOf, type AssetShaLookup, type BaseSnapshot, type SnapshotMedia, type TimelineAudioInput } from './snapshot'

export type RecheckState = 'changed' | 'unchanged' | 'unverifiable'

export interface RecheckResult {
  state: RecheckState
  note: string
}

export interface RecheckInput {
  category: ReviewCategory
  scopeKind: ReviewScopeKind
  shotId: string | null
  startMs: number | null
  endMs: number | null
  snapshot: BaseSnapshot
  timeline: FinalTimeline | null
  audio: TimelineAudioInput | null
  assetSha: AssetShaLookup
}

interface Window {
  startMs: number
  endMs: number
}

function windowOf(p: RecheckInput): Window | null {
  if (!p.timeline) return null
  if (p.scopeKind === 'shot') {
    if (!p.shotId) return null
    return shotRange(p.timeline, p.shotId)
  }
  if (p.scopeKind === 'time') {
    if (p.startMs === null || p.endMs === null) return null
    return { startMs: p.startMs, endMs: p.endMs }
  }
  return { startMs: 0, endMs: p.timeline.totalMs }
}

/** time 范围在新成片已越界（成片变短）→ 不做时间重映射，按 unverifiable 诚实降级（spec §4.6） */
function windowOutOfRange(p: RecheckInput): boolean {
  return (
    p.scopeKind === 'time' &&
    p.timeline !== null &&
    (p.startMs === null || p.endMs === null || p.endMs > p.timeline.totalMs)
  )
}

const mediaDiff = (
  base: SnapshotMedia,
  cur: SnapshotMedia,
): 'same' | 'diff' | 'sha-missing' => {
  if (base.assetId !== cur.assetId) return 'diff'
  if (base.sha256 && cur.sha256) return base.sha256 === cur.sha256 ? 'same' : 'diff'
  return 'sha-missing'
}

function fmtId(id: number | null): string {
  return id === null ? '—' : `#${id}`
}

/** 媒体类（visual/prop/lip_sync） */
function recheckMedia(p: RecheckInput): RecheckResult {
  if (!p.timeline) return { state: 'unverifiable', note: '当前成片缺少时间轴快照，无法比对镜头媒体' }
  if (windowOutOfRange(p)) return { state: 'unverifiable', note: '原时间窗超出当前成片，无法比对（需人工看片）' }
  const win = windowOf(p)
  if (!win) return { state: 'unverifiable', note: '原范围无法在当前成片重锚定' }

  if (p.scopeKind === 'shot') {
    const base = p.snapshot.shot
    if (!base) return { state: 'unverifiable', note: '基准快照未记录该镜媒体' }
    const seg = p.timeline.segments.find((s) => s.shotId === p.shotId)
    if (!seg) return { state: 'unverifiable', note: `镜头 ${p.shotId ?? ''} 已不在当前成片` }
    const cur: SnapshotMedia = { shotId: seg.shotId, assetId: seg.assetId, sha256: p.assetSha(seg.assetId), startMs: seg.startMs, endMs: seg.endMs, durMs: seg.durMs }
    const d = mediaDiff(base, cur)
    if (d === 'diff') return { state: 'changed', note: `该镜媒体已更换：${fmtId(base.assetId)} → ${fmtId(cur.assetId)}（是否修好仍需人看）` }
    if (d === 'sha-missing') return { state: 'unchanged', note: '该镜 assetId 未变；sha 不可用，仅比对 assetId' }
    return { state: 'unchanged', note: '该镜媒体未更换（是否修好仍需人看）' }
  }

  const baseSegs = p.snapshot.segments
  if (!baseSegs.length) return { state: 'unverifiable', note: '基准快照未记录范围内镜头媒体' }
  const curSegs = p.timeline.segments.filter((s) => s.endMs > win.startMs && s.startMs < win.endMs)
  const suffix = p.snapshot.segmentsTruncated ? '（基准段数超限，仅比较已存段）' : ''

  const curByShot = new Map(curSegs.map((s) => [s.shotId ?? `#${s.assetId}`, s]))
  let shaMissing = false
  const diffs: string[] = []
  for (const b of baseSegs) {
    const key = b.shotId ?? `#${b.assetId}`
    const c = curByShot.get(key)
    if (!c) {
      diffs.push(`${b.shotId ?? '该段'} 已不在当前窗口`)
      continue
    }
    const d = mediaDiff(b, { shotId: c.shotId, assetId: c.assetId, sha256: p.assetSha(c.assetId), startMs: c.startMs, endMs: c.endMs, durMs: c.durMs })
    if (d === 'diff') diffs.push(`${key}：${fmtId(b.assetId)} → ${fmtId(c.assetId)}`)
    if (d === 'sha-missing') shaMissing = true
  }
  if (!p.snapshot.segmentsTruncated && curSegs.length > baseSegs.length) {
    diffs.push('窗口内新增镜头段')
  }
  if (diffs.length) return { state: 'changed', note: `范围内媒体已变化：${diffs.join('；')}${suffix}` }
  if (shaMissing) return { state: 'unchanged', note: `范围内 assetId 未变；部分 sha 不可用${suffix}` }
  return { state: 'unchanged', note: `范围内媒体未更换${suffix}` }
}

/** 字幕 */
function recheckSubtitle(p: RecheckInput): RecheckResult {
  if (!p.timeline) return { state: 'unverifiable', note: '当前成片缺少时间轴快照，无法比对字幕' }
  if (windowOutOfRange(p)) return { state: 'unverifiable', note: '原时间窗超出当前成片，无法比对（需人工看片）' }
  const win = windowOf(p)
  if (!win) return { state: 'unverifiable', note: '原范围无法在当前成片重锚定' }
  const cur = p.timeline.cues
    .filter((c) => c.endMs > win.startMs && c.startMs < win.endMs)
    .map((c) => ({ id: c.id, startMs: c.startMs, endMs: c.endMs, text: c.text }))

  if (p.snapshot.cuesTruncated) {
    const digest = cuesDigestOf(cur)
    const same = p.snapshot.cuesDigest !== null && digest === p.snapshot.cuesDigest
    return {
      state: same ? 'unchanged' : 'changed',
      note: same
        ? `字幕摘要一致（窗口 ${p.snapshot.cuesCount} 条，未展开逐条）`
        : `字幕摘要变化（基准 ${p.snapshot.cuesCount} 条 → 当前 ${cur.length} 条，未展开逐条）`,
    }
  }

  const baseCues = p.snapshot.cues
  if (!baseCues.length && !cur.length) return { state: 'unchanged', note: '范围内无字幕可比' }
  if (!baseCues.length || !cur.length) {
    return { state: 'changed', note: `字幕增删：基准 ${baseCues.length} 条 → 当前 ${cur.length} 条` }
  }
  const baseKey = [...baseCues]
    .sort((a, b) => a.startMs - b.startMs || a.endMs - b.endMs || a.text.localeCompare(b.text))
    .map((c) => [c.startMs, c.endMs, c.text])
  const curKey = [...cur]
    .sort((a, b) => a.startMs - b.startMs || a.endMs - b.endMs || a.text.localeCompare(b.text))
    .map((c) => [c.startMs, c.endMs, c.text])
  const same = JSON.stringify(baseKey) === JSON.stringify(curKey)
  return {
    state: same ? 'unchanged' : 'changed',
    note: same
      ? `范围内字幕文本/起止未变（${cur.length} 条）`
      : `范围内字幕文本/起止已变化（基准 ${baseCues.length} 条 → 当前 ${cur.length} 条）`,
  }
}

/** 音频：只读 timeline.bgm/sfx（实际生效值），shot 范围只比该镜 SFX + BGM */
function recheckAudio(p: RecheckInput): RecheckResult {
  const base = p.snapshot.audio
  if (!base) return { state: 'unverifiable', note: '基准快照未记录音频证据' }
  if (!p.audio) return { state: 'unverifiable', note: '当前成片缺少音频时间轴证据' }
  const shotId = p.scopeKind === 'shot' ? p.shotId : null
  const baseSfx = shotId ? base.sfx.filter((s) => s.shotId === shotId) : base.sfx
  const curSfx = shotId ? p.audio.sfx.filter((s) => s.shotId === shotId) : p.audio.sfx

  const diffs: string[] = []
  if ((base.bgm?.assetId ?? null) !== (p.audio.bgm?.assetId ?? null)) {
    diffs.push(`BGM 换绑 ${fmtId(base.bgm?.assetId ?? null)} → ${fmtId(p.audio.bgm?.assetId ?? null)}`)
  } else if (base.bgm && p.audio.bgm) {
    if (base.bgm.sha256 && p.assetSha(base.bgm.assetId) && base.bgm.sha256 !== p.assetSha(base.bgm.assetId)) {
      diffs.push('BGM 内容变化')
    }
    if ((base.bgm.volume ?? null) !== (p.audio.bgm.volume ?? null)) diffs.push(`BGM 音量 ${base.bgm.volume} → ${p.audio.bgm.volume}`)
    if ((base.bgm.fadeSec ?? null) !== (p.audio.bgm.fadeSec ?? null)) diffs.push(`BGM 淡入 ${base.bgm.fadeSec} → ${p.audio.bgm.fadeSec}`)
  }

  const norm = (xs: Array<{ shotId: string | null; assetId: number | null }>) =>
    xs.map((s) => `${s.shotId ?? '*'}:${s.assetId ?? '-'}`).sort()
  if (JSON.stringify(norm(baseSfx)) !== JSON.stringify(norm(curSfx))) diffs.push('SFX 绑定变化')

  if (diffs.length) return { state: 'changed', note: `音频证据已变化：${diffs.join('；')}` }
  return { state: 'unchanged', note: '音频绑定/音量/淡入未变（听感仍需人耳确认）' }
}

/** 节奏：总长 + 范围内段长 */
function recheckPacing(p: RecheckInput): RecheckResult {
  if (!p.timeline) return { state: 'unverifiable', note: '当前成片缺少时间轴快照，无法比对节奏' }
  if (windowOutOfRange(p)) return { state: 'unverifiable', note: '原时间窗超出当前成片，无法比对（需人工看片）' }
  const diffs: string[] = []
  if (p.timeline.totalMs !== p.snapshot.totalMs) {
    diffs.push(`总长 ${p.snapshot.totalMs}ms → ${p.timeline.totalMs}ms`)
  }
  const win = windowOf(p)
  if (win) {
    const baseDur = new Map(p.snapshot.segments.map((s) => [s.shotId ?? `#${s.assetId}`, s.durMs]))
    const curSegs = p.timeline.segments.filter((s) => s.endMs > win.startMs && s.startMs < win.endMs)
    for (const s of curSegs) {
      const k = s.shotId ?? `#${s.assetId}`
      const b = baseDur.get(k)
      if (b !== undefined && b !== s.durMs) diffs.push(`${k} 段长 ${b}ms → ${s.durMs}ms`)
      if (b === undefined) diffs.push(`${k} 为新增段`)
    }
  }
  if (diffs.length) return { state: 'changed', note: `节奏证据已变化：${diffs.join('；')}` }
  return { state: 'unchanged', note: '总长与范围内段长未变' }
}

export function recheckScope(p: RecheckInput): RecheckResult {
  if (p.category === 'other') {
    return { state: 'unverifiable', note: '其他类问题无自动可比证据，需人工复核' }
  }
  if (p.category === 'subtitle') return recheckSubtitle(p)
  if (p.category === 'audio') return recheckAudio(p)
  if (p.category === 'pacing') return recheckPacing(p)
  return recheckMedia(p)
}
```

```ts
const isObj = (v: unknown): v is Record<string, unknown> =>
  !!v && typeof v === 'object' && !Array.isArray(v)

const strOrUndef = (v: unknown): string | undefined =>
  typeof v === 'string' ? v : undefined

function requireNote(v: unknown, field = 'note'): string {
  if (typeof v !== 'string') throw new ReviewError('note_required', `${field} 需为字符串`)
  const t = v.trim()
  if (!t) throw new ReviewError('note_required', `${field} 不能为空`)
  if (t.length > 2000) throw new ReviewError('note_required', `${field} 不能超过 2000 字符`)
  return t
}

function optionalNote(v: unknown): string | undefined {
  if (v === undefined || v === null) return undefined
  return requireNote(v)
}

function parseSeverity(v: unknown): ReviewSeverity {
  if (typeof v !== 'string' || !SEVERITIES.includes(v as ReviewSeverity)) {
    throw new ReviewError('bad_severity', `severity 需为 ${SEVERITIES.join('/')}`)
  }
  return v as ReviewSeverity
}

function parseCategory(v: unknown): ReviewCategory {
  if (typeof v !== 'string' || !CATEGORIES.includes(v as ReviewCategory)) {
    throw new ReviewError('bad_scope', `category 需为 ${CATEGORIES.join('/')}`)
  }
  return v as ReviewCategory
}

type ParsedScope =
  | { kind: 'shot'; shotId: string }
  | { kind: 'time'; startMs: number; endMs: number }
  | { kind: 'global' }

function parseScope(v: unknown): ParsedScope {
  if (!isObj(v)) throw new ReviewError('bad_scope', 'scope 需为对象')
  const kind = v['kind']
  if (kind === 'shot') {
    if (typeof v['shot_id'] !== 'string' || !v['shot_id']) throw new ReviewError('bad_scope', 'shot 范围需提供 shot_id')
    return { kind: 'shot', shotId: v['shot_id'] }
  }
  if (kind === 'time') {
    const startMs = v['start_ms']
    const endMs = v['end_ms']
    if (typeof startMs !== 'number' || typeof endMs !== 'number') throw new ReviewError('bad_scope', 'time 范围需提供 start_ms/end_ms')
    return { kind: 'time', startMs, endMs }
  }
  if (kind === 'global') return { kind: 'global' }
  throw new ReviewError('bad_scope', 'scope.kind 需为 shot/time/global')
}

async function loadShaMap(ids: Array<number | null | undefined>): Promise<Map<number, string | null>> {
  const uniq = [...new Set(ids.filter((x): x is number => typeof x === 'number' && Number.isInteger(x) && x > 0))]
  if (!uniq.length) return new Map()
  const rows = await db.select({ id: assets.id, sha256: assets.sha256 }).from(assets).where(inArray(assets.id, uniq))
  return new Map(rows.map((r) => [r.id, r.sha256]))
}

async function loadContext(runId: number, stepKey: string): Promise<ReviewContext> {
  const run = await loadRun(runId)
  const steps = await db.select().from(pipelineSteps).where(eq(pipelineSteps.runId, runId))
  const runAssets = await db.select().from(assets).where(eq(assets.runId, runId))
  const resolved = resolveReviewFinal(steps, runAssets, stepKey)
  const final = resolved ? runAssets.find((a) => a.id === resolved.assetId) ?? null : null
  const timeline = final ? parseFinalTimeline(final.params) : null
  const audio = final ? parseTimelineAudio(final.params) : null

  const ids: Array<number | null> = [
    final?.id ?? null,
    ...(timeline?.segments.map((s) => s.assetId) ?? []),
    audio?.bgm?.assetId ?? null,
    ...(audio?.sfx.map((s) => s.assetId) ?? []),
  ]
  const shaMap = await loadShaMap(ids)
  const assetSha: AssetShaLookup = (id) => (id === null ? null : shaMap.get(id) ?? null)
  const assetIds = new Set(runAssets.filter((a) => a.deletedAt == null).map((a) => a.id))

  const code: CapabilityCode =
    run.status === 'cancelled'
      ? 'run_cancelled'
      : ACTIVE_STATUSES.has(run.status)
        ? 'run_active'
        : !final
          ? 'no_final'
          : !timeline
            ? 'degraded_no_timeline'
            : 'ok'

  return {
    run,
    stepKey,
    final,
    timeline,
    audio,
    assetSha,
    assetIds,
    currentFinalId: final?.id ?? null,
    code,
    writable: code === 'ok' || code === 'degraded_no_timeline',
  }
}

function locationOf(row: ReviewFinding, ctx: ReviewContext): FindingLocation | null {
  if (row.scopeKind === 'global') return null
  if (row.scopeKind === 'shot') {
    if (ctx.timeline && row.shotId) {
      const r = shotRange(ctx.timeline, row.shotId)
      if (r) return { ...r, inRange: true }
    }
    return row.startMs !== null && row.endMs !== null
      ? { startMs: row.startMs, endMs: row.endMs, inRange: false }
      : null
  }
  const total = ctx.timeline?.totalMs ?? (ctx.final?.duration != null ? Math.round(ctx.final.duration * 1000) : null)
  const inRange = total !== null && row.endMs !== null && row.endMs <= total
  return row.startMs !== null && row.endMs !== null
    ? { startMs: row.startMs, endMs: row.endMs, inRange }
    : null
}

function canOf(row: ReviewFinding, ctx: ReviewContext): FindingCan {
  // spec §4.5：cancelled 只能只读 + patch；run_active 只拦 create/recheck（apply 后要能立即 link），
  // 其余推进动作不因执行中而锁死。
  const cancelled = ctx.code === 'run_cancelled'
  const active = ctx.code === 'run_active'
  const hasFinal = ctx.currentFinalId !== null
  const status = row.status as ReviewStatus
  const pending = status === 'pending_recheck'
  return {
    link: !cancelled && (status === 'open' || status === 'in_rework'),
    markRework: !cancelled && (status === 'open' || status === 'in_rework'),
    recheck:
      !cancelled &&
      !active &&
      hasFinal &&
      (status === 'open' || status === 'in_rework' || pending) &&
      (pending ? row.recheckFinalAssetId !== ctx.currentFinalId : row.baseFinalAssetId !== ctx.currentFinalId),
    verify: !cancelled && pending && hasFinal && row.recheckFinalAssetId === ctx.currentFinalId,
    reopen: !cancelled && (status === 'verified' || status === 'dismissed' || status === 'in_rework'),
    dismiss: !cancelled && (status === 'open' || status === 'in_rework' || pending),
  }
}

export async function createReviewFinding(
  runId: number,
  body: Record<string, unknown>,
): Promise<{ finding: FindingView; replayed: boolean }> {
  const stepKey = strOrUndef(body['step_key']) ?? DEFAULT_STEP_KEY
  const ctx = await loadContext(runId, stepKey)
  if (ctx.code === 'run_cancelled') throw new ReviewError('run_cancelled', '运行已取消，不能新增问题单')
  if (ctx.code === 'run_active') throw new ReviewError('run_active', '运行执行中不能新增问题单，请等待完成或取消')
  if (!ctx.final) throw new ReviewError('no_final', `步骤 ${stepKey} 当前没有成片`)

  const category = parseCategory(body['category'])
  const severity = parseSeverity(body['severity'])
  const note = requireNote(body['note'])
  const scope = parseScope(body['scope'])
  if (!categoryAllowsScope(category, scope.kind)) {
    throw new ReviewError('scope_category_mismatch', `类别 ${category} 不支持 ${scope.kind} 范围`)
  }

  let shotId: string | null = null
  let startMs: number | null = null
  let endMs: number | null = null
  if (scope.kind === 'shot') {
    if (!ctx.timeline) throw new ReviewError('no_timeline', '成片缺少时间轴快照，无法按镜头定位问题')
    const range = shotRange(ctx.timeline, scope.shotId)
    if (!range) throw new ReviewError('bad_scope', `镜头 ${scope.shotId} 不在当前成片时间轴`)
    shotId = scope.shotId
    startMs = range.startMs
    endMs = range.endMs
  } else if (scope.kind === 'time') {
    const fallback = ctx.final.duration != null ? Math.round(ctx.final.duration * 1000) : null
    const v = validateTimeRange(ctx.timeline, scope.startMs, scope.endMs, fallback)
    if (!v.ok) throw new ReviewError(v.code, v.message)
    startMs = scope.startMs
    endMs = scope.endMs
  }

  const clientKey = strOrUndef(body['client_key']) ?? null
  if (clientKey !== null && (clientKey.length === 0 || clientKey.length > 100)) {
    throw new ReviewError('bad_scope', 'client_key 需为 1-100 字符')
  }

  if (clientKey !== null) {
    const existing = (
      await db
        .select()
        .from(reviewFindings)
        .where(and(eq(reviewFindings.runId, runId), eq(reviewFindings.clientKey, clientKey)))
        .limit(1)
    )[0]
    if (existing) {
      const same =
        existing.stepKey === stepKey &&
        existing.category === category &&
        existing.severity === severity &&
        existing.note === note &&
        existing.scopeKind === scope.kind &&
        existing.shotId === shotId &&
        existing.startMs === startMs &&
        existing.endMs === endMs
      if (!same) throw new ReviewError('client_key_conflict', 'client_key 已被不同内容占用')
      return { finding: toView(existing, ctx), replayed: true }
    }
  }

  const snapshot = buildBaseSnapshot({
    stepKey,
    finalAssetId: ctx.final.id,
    finalSha256: ctx.final.sha256,
    timeline: ctx.timeline,
    scope,
    audio: ctx.audio,
    assetSha: ctx.assetSha,
  })
  const now = Date.now()
  const values = {
    projectId: ctx.run.projectId,
    runId,
    stepKey,
    baseFinalAssetId: ctx.final.id,
    clientKey,
    scopeKind: scope.kind,
    shotId,
    startMs,
    endMs,
    category,
    severity,
    note,
    status: 'open',
    baseSnapshot: JSON.stringify(snapshot),
    createdAt: now,
    updatedAt: now,
  }
  try {
    const row = (await db.insert(reviewFindings).values(values).returning())[0]!
    return { finding: toView(row, ctx), replayed: false }
  } catch (err) {
    // 并发首写碰撞：唯一索引兜底后复查，同载荷回放 / 异载荷冲突
    if (clientKey === null) throw err
    const raced = (
      await db
        .select()
        .from(reviewFindings)
        .where(and(eq(reviewFindings.runId, runId), eq(reviewFindings.clientKey, clientKey)))
        .limit(1)
    )[0]
    if (!raced) throw err
    const same =
      raced.stepKey === stepKey &&
      raced.category === category &&
      raced.severity === severity &&
      raced.note === note &&
      raced.scopeKind === scope.kind &&
      raced.shotId === shotId &&
      raced.startMs === startMs &&
      raced.endMs === endMs
    if (!same) throw new ReviewError('client_key_conflict', 'client_key 已被不同内容占用')
    return { finding: toView(raced, ctx), replayed: true }
  }
}

export async function patchReviewFinding(
  runId: number,
  fid: number,
  body: Record<string, unknown>,
): Promise<FindingView> {
  const row = await getFinding(runId, fid)
  const patch: Record<string, unknown> = {}
  if (body['note'] !== undefined) patch['note'] = requireNote(body['note'])
  if (body['severity'] !== undefined) patch['severity'] = parseSeverity(body['severity'])
  if (!Object.keys(patch).length) return toView(row, await loadContext(runId, row.stepKey))
  const updated = await save(fid, patch)
  return toView(updated, await loadContext(runId, row.stepKey))
}

export async function linkFinding(
  runId: number,
  fid: number,
  body: Record<string, unknown>,
): Promise<FindingView> {
  const row = await getFinding(runId, fid)
  const ctx = await loadContext(runId, row.stepKey)
  if (ctx.code === 'run_cancelled') throw new ReviewError('run_cancelled', '运行已取消，不能关联返修')
  if (row.status !== 'open' && row.status !== 'in_rework') {
    throw new ReviewError('bad_transition', `状态 ${row.status} 不能关联返修`)
  }
  const requestId = body['request_id']
  if (typeof requestId !== 'string' || !requestId) throw new ReviewError('request_not_found', 'request_id 必填')
  const req = (await db.select().from(reworkRequests).where(eq(reworkRequests.id, requestId)).limit(1))[0]
  if (!req) throw new ReviewError('request_not_found', `返修请求 ${requestId} 不存在`)
  if (req.runId !== runId) throw new ReviewError('request_run_mismatch', '返修请求不属于该运行')
  if (req.stepKey !== row.stepKey) throw new ReviewError('bad_transition', `返修请求属于步骤 ${req.stepKey}，与问题单步骤 ${row.stepKey} 不一致`)
  if (req.state !== 'applied') throw new ReviewError('request_not_applied', `返修请求尚未应用（${req.state}）`)
  const updated = await save(fid, { status: 'in_rework', reworkRequestId: requestId })
  return toView(updated, ctx)
}

export async function recheckFinding(runId: number, fid: number): Promise<FindingView> {
  const row = await getFinding(runId, fid)
  const ctx = await loadContext(runId, row.stepKey)
  if (ctx.code === 'run_cancelled') throw new ReviewError('run_cancelled', '运行已取消，不能复检')
  if (ctx.code === 'run_active') throw new ReviewError('run_active', '运行执行中，待新成片产出后再复检')
  if (!ctx.final) throw new ReviewError('no_final', `步骤 ${row.stepKey} 当前没有成片`)
  if (!['open', 'in_rework', 'pending_recheck'].includes(row.status)) {
    throw new ReviewError('bad_transition', `状态 ${row.status} 不能复检`)
  }
  const ref = row.status === 'pending_recheck' ? row.recheckFinalAssetId : row.baseFinalAssetId
  if (ctx.currentFinalId === ref) throw new ReviewError('final_unchanged', '成片未变化，无需复检')

  const snapshot = parseBaseSnapshot(row.baseSnapshot)
  const result = snapshot
    ? recheckScope({
        category: row.category as ReviewCategory,
        scopeKind: row.scopeKind as ReviewScopeKind,
        shotId: row.shotId,
        startMs: row.startMs,
        endMs: row.endMs,
        snapshot,
        timeline: ctx.timeline,
        audio: ctx.audio,
        assetSha: ctx.assetSha,
      })
    : { state: 'unverifiable' as const, note: '基准快照不可读，无法自动比对（需人工看片）' }

  const updated = await save(fid, {
    status: 'pending_recheck',
    recheckState: result.state,
    recheckFinalAssetId: ctx.currentFinalId,
    recheckNote: result.note,
    recheckAt: Date.now(),
  })
  return toView(updated, ctx)
}

export async function verifyFinding(
  runId: number,
  fid: number,
  body: Record<string, unknown>,
): Promise<FindingView> {
  const row = await getFinding(runId, fid)
  const ctx = await loadContext(runId, row.stepKey)
  if (ctx.code === 'run_cancelled') throw new ReviewError('run_cancelled', '运行已取消，不能终判')
  if (row.status !== 'pending_recheck') throw new ReviewError('bad_transition', `状态 ${row.status} 不是待复检`)
  const result = body['result']
  if (result !== 'verified' && result !== 'reopened') {
    throw new ReviewError('bad_result', 'result 需为 verified 或 reopened')
  }
  const finalAssetId = body['final_asset_id']
  if (typeof finalAssetId !== 'number' || !Number.isInteger(finalAssetId) || finalAssetId <= 0) {
    throw new ReviewError('bad_scope', 'final_asset_id 需为正整数')
  }
  if (ctx.currentFinalId !== row.recheckFinalAssetId) {
    throw new ReviewError('stale_recheck', '成片在复检后已再次变化，请重新复检')
  }
  if (finalAssetId !== ctx.currentFinalId) {
    throw new ReviewError('stale_final', '提交的成片与当前成片不一致，请刷新后重试')
  }
  const note = optionalNote(body['note'])
  const now = Date.now()
  const patch =
    result === 'verified'
      ? { status: 'verified', resolvedAt: now, resolutionNote: note ?? null }
      : { status: 'open', resolvedAt: null, resolutionNote: note ?? null }
  const updated = await save(fid, patch)
  return toView(updated, ctx)
}
```

```ts
export const reviewRoutes = new Hono()

const STATUS: Record<string, number> = {
  run_not_found: 404,
  finding_not_found: 404,
  bad_scope: 400,
  scope_category_mismatch: 400,
  note_required: 400,
  bad_severity: 400,
  bad_result: 400,
  client_key_conflict: 400,
  run_active: 409,

  stale_recheck: 409,
  bad_transition: 409,

function failed(err: unknown): never {
  if (err instanceof ReviewError) {

  if (!body || typeof body !== 'object' || Array.isArray(body)) {
    throw new HttpError(400, 'bad_scope', '请求体需为 JSON 对象')
  }
  return body as Record<string, unknown>
}
```

```ts
/**
 * M69 审片面板状态：加载、动作、当前播放时间与镜头定位。
 * 服务端投影为准；本 composable 只做 UI 态（loading/acting/notice）与半开区间当前镜头。
 */
import { computed, ref, type Ref } from 'vue'
import { findingsApi } from '../../../lib/api/findings'
import { ApiError } from '../../../lib/api/core'
import type { FindingsList, FindingView, ReviewScopeKind } from '../../../lib/types/findings'

export function useRunFindings(runId: Ref<number>, stepKey = 'compose') {
  const data = ref<FindingsList | null>(null)
  const loading = ref(false)
  const acting = ref(false)
  const hidden = ref(false)
  const errText = ref('')
  const notice = ref('')
  const currentMs = ref(0)

  const formatMs = (ms: number): string => {
    const total = Math.max(0, Math.round(ms))
    const s = Math.floor(total / 1000)
    const m = Math.floor(s / 60)
    return `${String(m).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}.${Math.floor((total % 1000) / 100)}`
  }

  const segments = computed(() => data.value?.timeline?.segments ?? [])
  const totalMs = computed(() => data.value?.timeline?.totalMs ?? data.value?.final?.durationMs ?? 0)
  const finalAssetId = computed(() => data.value?.final?.assetId ?? null)

  /** 与后端 locateAt 同口径：半开区间，边界取后一段 */
  const currentShot = computed(() => {
    const tl = data.value?.timeline
    if (!tl) return null
    const at = currentMs.value
    if (at < tl.introMs || at >= tl.totalMs) return null
    return tl.segments.find((s) => at >= s.startMs && at < s.endMs) ?? null
  })

  const findings = computed(() => data.value?.findings ?? [])
  const sortedFindings = computed(() => {
    const rank = (s: FindingView['status']): number =>
      s === 'open' || s === 'in_rework' ? 0 : s === 'pending_recheck' ? 1 : 2
    const sev = (s: FindingView['severity']): number => (s === 'blocker' ? 0 : s === 'major' ? 1 : 2)
    return [...findings.value].sort(
      (a, b) => rank(a.status) - rank(b.status) || sev(a.severity) - sev(b.severity) || a.createdAt - b.createdAt,
    )
  })

  async function load(): Promise<void> {
    loading.value = true
    errText.value = ''
    try {
      data.value = await findingsApi.list(runId.value, stepKey)
      // 无成片且无历史问题单：整体隐藏，不打扰（与 QC 面板同策略）
      hidden.value = data.value.capability.code === 'no_final' && data.value.findings.length === 0
    } catch (e) {
      if (e instanceof ApiError && e.code === 'run_not_found') hidden.value = true
      else errText.value = e instanceof ApiError ? e.message : '读取审片问题失败'
    } finally {
      loading.value = false
    }
  }

  function setMs(ms: number): void {
    currentMs.value = Number.isFinite(ms) ? ms : 0
  }

  function seekTo(ms: number, video: HTMLVideoElement | null): void {
    setMs(ms)
    if (video) video.currentTime = Math.max(0, ms / 1000)
  }

  async function act(fn: () => Promise<unknown>, okText: string): Promise<void> {
    acting.value = true
    notice.value = ''
    errText.value = ''
    try {
      await fn()
      notice.value = okText
      await load()
    } catch (e) {
      errText.value = e instanceof ApiError ? e.message : '操作失败'
    } finally {
      acting.value = false
    }
  }

  function allowedScopes(category: FindingView['category']): ReviewScopeKind[] {
    if (category === 'lip_sync') return ['shot']
    if (category === 'pacing') return ['time', 'global']
    return ['shot', 'time', 'global']
  }

  return {
    data, loading, acting, hidden, errText, notice, currentMs,
    formatMs, segments, totalMs, finalAssetId, currentShot,
    findings, sortedFindings, load, setMs, seekTo, act, allowedScopes,
  }
}
```

```vue
    notice.value = '未找到镜头卡锚点，请在镜头工作台手动定位'
  }
  seekTo(f.currentLocation.startMs, videoEl.value)
}

function onTimeUpdate(e: Event): void {
  const el = e.target as HTMLVideoElement | null
  if (el) setMs(el.currentTime * 1000)
}

function onAppliedModal(requestId: string): void {
  const id = modal.value?.findingId
  const f = id ? sortedFindings.value.find((x) => x.id === id) : undefined
  if (f) void linkRequest(f, requestId)
}

function jumpToMark(): void {
  const loc = sortedFindings.value.find((f) => f.currentLocation)?.currentLocation
  if (loc) seekTo(loc.startMs, videoEl.value)
}

function verify(f: FindingView, result: 'verified' | 'reopened'): void {
  if (finalAssetId.value === null) return
  const note = window.prompt(result === 'verified' ? '确认已修复（可选备注）' : '仍未修复（可选备注）') ?? undefined
  void act(
    () => findingsApi.verify(props.runId, f.id, { result, final_asset_id: finalAssetId.value!, note }),
    result === 'verified' ? '已确认修复' : '已重新打开',
  )
}

function dismiss(f: FindingView): void {
  const note = window.prompt('忽略原因（可选）') ?? undefined
  void act(() => findingsApi.dismiss(props.runId, f.id, note), '已忽略')
}

async function linkRequest(f: FindingView, requestId: string): Promise<void> {
  await act(() => findingsApi.link(props.runId, f.id, requestId), '已关联返修')
  modal.value = null
}
```

- 真实成片的主观质量（视觉一致/口型/听感）无法由探针验证，只能人工看片；本计划的验证是静态+合成夹具，不代表真实出片质量。
- `time` 范围在新成片发生位移时只报 `unverifiable`/`changed`，不做时间重映射（spec §七已声明）。
- 多合成步按 `step_key` 隔离，首版面板默认 `compose`，不做同页多面板。
- spec §4.8 第 8 条要求 `ShotCard` 加 `data-step-key`，但 `ShotBoardShot` 无 `stepKey` 字段；本计划只加 `id="shot-<shotId>"` 与 `data-shot-id`，不从卡片反推步骤（避免造第二真源）。
