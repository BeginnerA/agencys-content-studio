export function foldEntries(entries: { ref: EntryRef; doc: EntryDoc }[]) {
  const slot = (i: Item) => JSON.stringify([i.channel, i.channel === 'knowledge' ? i.knower : null, i.topicKey])
  const heads = new Map<string, { ref: ItemRef; op: Item['operation'] }>()
  const ordered = entries.toSorted((a, b) => a.ref.unitOrdinal - b.ref.unitOrdinal)
  for (const entry of ordered) {
    const touched = new Set<string>()
    for (const item of entry.doc.items.filter(i => i.included)) {
      const key = slot(item)
      if (touched.has(key)) throw new ContinuityError('continuity_conflict', 409)
      touched.add(key)
      const previous = heads.get(key)
      if (!previous && (item.previous !== null || item.operation !== 'set')) throw new ContinuityError('continuity_dependency_invalid', 409)
      if (previous && canonicalHash(previous.ref) !== canonicalHash(item.previous)) throw new ContinuityError('continuity_conflict', 409)
      heads.set(key, { ref: { ...entry.ref, itemId: item.id }, op: item.operation })
    }
  }
  const refs = (op: Item['operation']) => Array.from(heads.values()).filter(v => v.op === op).map(v => v.ref)
  return { current: refs('set'), closed: refs('close'), retracted: refs('retract') }
}
